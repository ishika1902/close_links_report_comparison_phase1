from api_client import call_query
from transformer import build_dataframe
from comparator import compare_data
from logger import write_log
from datetime import date


# AUTOMATIC SAFE 10TH LOGIC
def get_reference_dates():
    today = date.today()

    if today.day > 10:
        current_ref = date(today.year, today.month, 10)
    else:
        if today.month == 1:
            current_ref = date(today.year - 1, 12, 10)
        else:
            current_ref = date(today.year, today.month - 1, 10)

    if current_ref.month == 1:
        prev_ref = date(current_ref.year - 1, 12, 10)
    else:
        prev_ref = date(current_ref.year, current_ref.month - 1, 10)

    return prev_ref.isoformat(), current_ref.isoformat()


def main():

    previous_reference_date, current_reference_date = get_reference_dates()

    access_token = "YOUR_ACCESS_TOKEN"

    print("Previous Ref Date:", previous_reference_date)
    print("Current Ref Date:", current_reference_date)

    prev_data = call_query(previous_reference_date, access_token)
    curr_data = call_query(current_reference_date, access_token)

    prev_df = build_dataframe(prev_data)
    curr_df = build_dataframe(curr_data)

    added, deleted, modified = compare_data(prev_df, curr_df)

    prev_month = previous_reference_date[5:7]
    curr_month = current_reference_date[5:7]
    year = current_reference_date[:4]

    log_file = write_log(added, deleted, modified, prev_month, curr_month, year)

    print(f"Log file generated: {log_file}")


if __name__ == "__main__":
    main()
