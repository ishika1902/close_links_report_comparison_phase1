import os
from datetime import datetime
from dateutil.relativedelta import relativedelta

from api_client import call_query
from transformer import build_dataframe
from comparator import compare_data
from logger import generate_log_file


# ---------------------------------------------------
# Query Names (EXACT names from API)
# ---------------------------------------------------

FCA_QUERY = "Close Links Report - for FCA Submission"
STATUS_QUERY = "Entity list with Statuses"


# ---------------------------------------------------
# Reference Date Calculation
# ---------------------------------------------------

def get_reference_dates():
    today = datetime.today()

    # Current = 10th of current month
    current = today.replace(day=10)

    # Previous = 10th of previous month
    previous = (current - relativedelta(months=1)).replace(day=10)

    return previous.strftime("%Y-%m-%d"), current.strftime("%Y-%m-%d")


# ---------------------------------------------------
# Main Execution
# ---------------------------------------------------

def main():

    previous_reference_date, current_reference_date = get_reference_dates()

    print(f"Previous Ref Date: {previous_reference_date}")
    print(f"Current Ref Date: {current_reference_date}")

    # ---------------------------------------------------
    # 1️⃣ Call FCA Query Twice (Prev + Current)
    # ---------------------------------------------------

    print("Fetching Previous FCA Data...")
    prev_raw = call_query(FCA_QUERY, previous_reference_date)

    print("Fetching Current FCA Data...")
    curr_raw = call_query(FCA_QUERY, current_reference_date)

    # ---------------------------------------------------
    # 2️⃣ Call Status Query Once
    # ---------------------------------------------------

    print("Fetching Status Data...")
    status_raw = call_query(STATUS_QUERY, current_reference_date)

    # ---------------------------------------------------
    # 3️⃣ Transform JSON → DataFrames
    # ---------------------------------------------------

    prev_df = build_dataframe(prev_raw)
    curr_df = build_dataframe(curr_raw)
    status_df = build_dataframe(status_raw)

    # ---------------------------------------------------
    # 4️⃣ Compare Data
    # ---------------------------------------------------

    comparison_df = compare_data(prev_df, curr_df, status_df)

    # ---------------------------------------------------
    # 5️⃣ Generate Log File Name
    # ---------------------------------------------------

    prev_date_obj = datetime.strptime(previous_reference_date, "%Y-%m-%d")
    curr_date_obj = datetime.strptime(current_reference_date, "%Y-%m-%d")

    prev_month_name = prev_date_obj.strftime("%B")
    curr_month_name = curr_date_obj.strftime("%B")
    year = curr_date_obj.strftime("%Y")

    file_name = f"FCA_comparison_log_{prev_month_name}_vs_{curr_month_name}_{year}.csv"

    # ---------------------------------------------------
    # 6️⃣ Generate Log File
    # ---------------------------------------------------

    generate_log_file(comparison_df, file_name)

    print(f"\nLog file generated successfully: {file_name}")


# ---------------------------------------------------

if __name__ == "__main__":
    main()
