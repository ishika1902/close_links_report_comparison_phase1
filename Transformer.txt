import pandas as pd
import re


def clean_value(val):
    if val is None:
        return ""
    return str(val).strip()


def normalize_address(address):
    address = re.sub(r',+', ',', address)
    address = re.sub(r'\s+', ' ', address)
    return address.strip()


def build_registered_address(item):
    parts = [
        clean_value(item.get("SubsidiaryEntityRel.CurrentMainRegisteredAddress_SNum")),
        clean_value(item.get("SubsidiaryEntityRel.CurrentMainRegisteredAddress_SN")),
        clean_value(item.get("SubsidiaryEntityRel.CurrentMainRegisteredAddress_NativeSN")),
        clean_value(item.get("SubsidiaryEntityRel.CurrentMainRegisteredAddress_E")),
        clean_value(item.get("SubsidiaryEntityRel.CurrentMainRegisteredAddress_NativeE")),
        clean_value(item.get("SubsidiaryEntityRel.CurrentMainRegisteredAddress_City")),
        clean_value(item.get("SubsidiaryEntityRel.CurrentMainRegisteredAddress_NativeCity")),
        clean_value(item.get("SubsidiaryEntityRel.CurrentMainRegisteredAddress_Country")),
    ]

    parts = [p for p in parts if p]
    address = ", ".join(parts)

    return normalize_address(address)


def build_dataframe(api_data):

    rows = []

    if isinstance(api_data, dict):
        api_data = api_data.get("value", [])

    for item in api_data:
        row = {
            "CorporationID": clean_value(item.get("CorporationID")),
            "Subsidiary": clean_value(item.get("Subsidiary")),
            "Country ISO-2 Code": clean_value(item.get("Country ISO-2 Code")),
            "Entity Registration Number": clean_value(item.get("Entity Registration Number")),
            "Registered Address": build_registered_address(item),
        }
        rows.append(row)

    return pd.DataFrame(rows)


def build_status_map(status_data):

    status_map = {}

    if isinstance(status_data, dict):
        status_data = status_data.get("value", [])

    for item in status_data:
        entity_id = clean_value(item.get("EntityID"))

        status_map[entity_id] = {
            "Status": clean_value(item.get("Status")),
            "DateModified": clean_value(item.get("DateModified"))
        }

    return status_map
