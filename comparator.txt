import pandas as pd


def compare_snapshots(old_df, new_df, status_map):

    result_rows = []

    old_dict = old_df.set_index("CorporationID").to_dict("index")
    new_dict = new_df.set_index("CorporationID").to_dict("index")

    old_ids = set(old_dict.keys())
    new_ids = set(new_dict.keys())

    added = new_ids - old_ids
    deleted = old_ids - new_ids
    common = old_ids & new_ids

    # ADDED
    for cid in added:
        row = new_dict[cid]
        status_info = status_map.get(cid, {})
        result_rows.append({
            "Section": "ADDED",
            "Change Type": "ADDED",
            "Version": "",
            **row,
            "Changed Fields": "",
            **status_info
        })

    # DELETED
    for cid in deleted:
        row = old_dict[cid]
        status_info = status_map.get(cid, {})
        result_rows.append({
            "Section": "DELETED",
            "Change Type": "DELETED",
            "Version": "",
            **row,
            "Changed Fields": "",
            **status_info
        })

    # MODIFIED
    for cid in common:
        old_row = old_dict[cid]
        new_row = new_dict[cid]

        changed_fields = []

        for col in [
            "Subsidiary",
            "Country ISO-2 Code",
            "Entity Registration Number",
            "Registered Address"
        ]:
            if old_row[col] != new_row[col]:
                changed_fields.append(col)

        if changed_fields:
            status_info = status_map.get(cid, {})

            result_rows.append({
                "Section": "MODIFIED",
                "Change Type": "MODIFIED",
                "Version": "OLD",
                **old_row,
                "Changed Fields": ", ".join(changed_fields),
                **status_info
            })

            result_rows.append({
                "Section": "MODIFIED",
                "Change Type": "MODIFIED",
                "Version": "NEW",
                **new_row,
                "Changed Fields": ", ".join(changed_fields),
                **status_info
            })

    return pd.DataFrame(result_rows)
