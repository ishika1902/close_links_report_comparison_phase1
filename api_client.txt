import requests
import json
from datetime import datetime
from dateutil.relativedelta import relativedelta


def getData():
    prod_url = "https://englobe-uat.hk.hsbc/ids/connect/token"
    payload = "xxx"

    headers = {
        'content-type': "application/x-www-form-urlencoded",
        'cache-control': "no-cache"
    }

    response_token = requests.request(
        "POST",
        prod_url,
        data=payload,
        headers=headers,
        verify=False
    )

    token = response_token.text.split(',')[0]
    str_len = len(token)
    access_token_server = token[16:str_len - 1]
    access_token_server = access_token_server[1:]
    auth_key = "bearer" + access_token_server

    return auth_key


def get_reference_dates():
    today = datetime.today()
    current = today.replace(day=10)
    previous = (current - relativedelta(months=1)).replace(day=10)

    return previous.strftime("%Y-%m-%d"), current.strftime("%Y-%m-%d")


def call_query(query_name, reference_date):

    prod_query_url = f"https://englobe-uat.hk.hsbc/api/api/queries/{query_name}"

    auth = getData()

    payload = json.dumps({
        "RefDate": reference_date   # ✅ CORRECT KEY
    })

    headers = {
        'authorization': auth,
        'cache-control': "no-cache",
        'Accept': 'application/json',
        'Content-Type': 'application/json'
    }

    response = requests.request(
        "POST",
        prod_query_url,
        data=payload,
        headers=headers,
        verify=False
    )

    return response.json()
