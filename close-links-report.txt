import pandas as pd
import requests
import os

# =========================
# CONFIGURATION
# =========================

PREVIOUS_FILE = "input/previous_month.xlsx"
CURRENT_FILE = "input/current_month.xlsx"
OUTPUT_FILE = "output/comparison_output.xlsx"

AUTH_URL = "https://your-auth-url"
MAIN_REPORT_API = "https://your-main-report-api-url"
STATUS_API = "https://your-status-api-url"

USERNAME = "your_username"
PASSWORD = "your_password"

# =========================
# AUTHENTICATION
# =========================

def get_auth_token():
    response = requests.post(
        AUTH_URL,
        json={"username": USERNAME, "password": PASSWORD}
    )
    response.raise_for_status()
    return response.json().get("token")

# =========================
# LOAD EXCEL
# =========================

def load_report(path):
    df = pd.read_excel(path, skiprows=3)
    df.columns = df.columns.str.strip()

    # Remove unwanted columns
    df = df.drop(columns=["Voting (%)", "Interest (%)"], errors="ignore")

    return df

# =========================
# FILL BLANK REGISTRATION FROM PREVIOUS MONTH
# =========================

def fill_blank_registration(curr_df, prev_df):
    for idx, row in curr_df.iterrows():
        if pd.isna(row.get("Entity Registration Number")):
            match = prev_df[
                (prev_df["Subsidiary"] == row["Subsidiary"]) &
                (prev_df["Country ISO-2 Code"] == row["Country ISO-2 Code"])
            ]
            if not match.empty:
                curr_df.at[idx, "Entity Registration Number"] = \
                    match.iloc[0]["Entity Registration Number"]

    return curr_df

# =========================
# FETCH FULL API DATASET
# =========================

def fetch_full_api_dataset(headers):
    response = requests.get(MAIN_REPORT_API, headers=headers)
    response.raise_for_status()
    return pd.DataFrame(response.json())

# =========================
# ENRICH EXCEL WITH CORPORATION ID
# =========================

def enrich_with_corporation_id(df, api_df):

    api_df.columns = api_df.columns.str.strip()

    # Build registration → CorporationID mapping
    reg_map = dict(
        zip(api_df["Entity Registration Number"], api_df["CorporationID"])
    )

    # Map CorporationID
    df["CorporationID"] = df["Entity Registration Number"].map(reg_map)

    # Fallback matching: Subsidiary + Country
    for idx, row in df[df["CorporationID"].isna()].iterrows():
        fallback = api_df[
            (api_df["Subsidiary"] == row["Subsidiary"]) &
            (api_df["Country ISO-2 Code"] == row["Country ISO-2 Code"])
        ]
        if not fallback.empty:
            df.at[idx, "CorporationID"] = fallback.iloc[0]["CorporationID"]

    return df

# =========================
# FETCH STATUS DATA
# =========================

def fetch_status_dataset(headers):
    response = requests.get(STATUS_API, headers=headers)
    response.raise_for_status()
    return pd.DataFrame(response.json())

# =========================
# COMPARE USING CORPORATION ID
# =========================

def compare_reports(prev_df, curr_df, status_df):

    comparison_columns = [
        "Subsidiary",
        "Country ISO-2 Code",
        "Entity Registration Number",
        "Registered Address"
    ]

    prev_df = prev_df.set_index("CorporationID")
    curr_df = curr_df.set_index("CorporationID")

    prev_ids = set(prev_df.index)
    curr_ids = set(curr_df.index)

    added_ids = curr_ids - prev_ids
    deleted_ids = prev_ids - curr_ids
    common_ids = prev_ids & curr_ids

    results = []

    # Added
    for cid in added_ids:
        row = curr_df.loc[cid]
        results.append({
            "CorporationID": cid,
            "ChangeType": "Added",
            **row.to_dict()
        })

    # Deleted
    for cid in deleted_ids:
        row = prev_df.loc[cid]
        status_row = status_df[status_df["CorporationID"] == cid]

        status = None
        when_modified = None

        if not status_row.empty:
            status = status_row.iloc[0].get("Status")
            when_modified = status_row.iloc[0].get("WhenModified")

        results.append({
            "CorporationID": cid,
            "ChangeType": "Deleted",
            "Status": status,
            "WhenModified": when_modified,
            **row.to_dict()
        })

    # Modified
    for cid in common_ids:
        prev_row = prev_df.loc[cid]
        curr_row = curr_df.loc[cid]

        for col in comparison_columns:
            if prev_row.get(col) != curr_row.get(col):
                status_row = status_df[status_df["CorporationID"] == cid]
                when_modified = None

                if not status_row.empty:
                    when_modified = status_row.iloc[0].get("WhenModified")

                results.append({
                    "CorporationID": cid,
                    "ChangeType": "Modified",
                    "ModifiedColumn": col,
                    "OldValue": prev_row.get(col),
                    "NewValue": curr_row.get(col),
                    "WhenModified": when_modified
                })

    return pd.DataFrame(results)

# =========================
# MAIN
# =========================

def main():

    os.makedirs("output", exist_ok=True)

    token = get_auth_token()
    headers = {"Authorization": f"Bearer {token}"}

    prev_df = load_report(PREVIOUS_FILE)
    curr_df = load_report(CURRENT_FILE)

    curr_df = fill_blank_registration(curr_df, prev_df)

    api_df = fetch_full_api_dataset(headers)

    prev_df = enrich_with_corporation_id(prev_df, api_df)
    curr_df = enrich_with_corporation_id(curr_df, api_df)

    # Safety check
    if prev_df["CorporationID"].isna().any() or \
       curr_df["CorporationID"].isna().any():
        raise ValueError("Some rows could not be mapped to CorporationID.")

    status_df = fetch_status_dataset(headers)

    comparison_df = compare_reports(prev_df, curr_df, status_df)

    comparison_df.to_excel(OUTPUT_FILE, index=False)

    print("Comparison completed successfully.")

if __name__ == "__main__":
    main()
