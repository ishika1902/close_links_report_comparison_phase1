import pandas as pd
import requests
import os

# =========================
# CONFIGURATION
# =========================

PREVIOUS_FILE = "input/previous_month.xlsx"
CURRENT_FILE = "input/current_month.xlsx"
OUTPUT_FILE = "output/comparison_output.xlsx"

MAIN_REPORT_API = "https://your-main-report-api-url"
STATUS_API = "https://your-status-api-url"

USERNAME = "your_username"
PASSWORD = "your_password"

# =========================
# AUTHENTICATION
# =========================

def get_auth_token():
    response = requests.post(
        "https://your-auth-url",
        json={"username": USERNAME, "password": PASSWORD}
    )
    return response.json().get("token")

# =========================
# LOAD EXCEL FILES
# =========================

def load_report(file_path):
    df = pd.read_excel(file_path, skiprows=3)

    # Remove unwanted columns
    df = df.drop(columns=["Voting (%)", "Interest (%)"], errors="ignore")

    # Standardize column names (strip spaces)
    df.columns = df.columns.str.strip()

    return df

# =========================
# FIX BLANK REGISTRATION NUMBERS
# =========================

def fetch_registration_from_api(entity_id, headers):
    response = requests.get(
        f"{MAIN_REPORT_API}/{entity_id}",
        headers=headers
    )

    if response.status_code == 200:
        data = response.json()
        return data.get("Entity Registration Number")
    return None

def fix_blank_registrations(df, headers):
    for index, row in df.iterrows():
        if pd.isna(row.get("Entity Registration Number")):
            entity_id = row.get("CorporationID")
            latest_reg = fetch_registration_from_api(entity_id, headers)
            if latest_reg:
                df.at[index, "Entity Registration Number"] = latest_reg
    return df

# =========================
# FETCH STATUS DATA
# =========================

def fetch_status_dataset(headers):
    response = requests.get(STATUS_API, headers=headers)
    if response.status_code == 200:
        return pd.DataFrame(response.json())
    return pd.DataFrame()

# =========================
# COMPARE REPORTS
# =========================

def compare_reports(prev_df, curr_df, status_df):

    comparison_columns = [
        "Subsidiary",
        "Country ISO-2 Code",
        "Entity Registration Number",
        "Registered Address"
    ]

    prev_df = prev_df.set_index("CorporationID")
    curr_df = curr_df.set_index("CorporationID")

    prev_ids = set(prev_df.index)
    curr_ids = set(curr_df.index)

    added_ids = curr_ids - prev_ids
    deleted_ids = prev_ids - curr_ids
    common_ids = prev_ids & curr_ids

    results = []

    # Added
    for cid in added_ids:
        row = curr_df.loc[cid]
        results.append({
            "CorporationID": cid,
            "ChangeType": "Added",
            **row.to_dict()
        })

    # Deleted
    for cid in deleted_ids:
        row = prev_df.loc[cid]
        status_row = status_df[status_df["CorporationID"] == cid]

        when_modified = None
        status = None

        if not status_row.empty:
            when_modified = status_row.iloc[0].get("WhenModified")
            status = status_row.iloc[0].get("Status")

        results.append({
            "CorporationID": cid,
            "ChangeType": "Deleted",
            "Status": status,
            "WhenModified": when_modified,
            **row.to_dict()
        })

    # Modified
    for cid in common_ids:
        prev_row = prev_df.loc[cid]
        curr_row = curr_df.loc[cid]

        for col in comparison_columns:
            if prev_row.get(col) != curr_row.get(col):
                status_row = status_df[status_df["CorporationID"] == cid]
                when_modified = None

                if not status_row.empty:
                    when_modified = status_row.iloc[0].get("WhenModified")

                results.append({
                    "CorporationID": cid,
                    "ChangeType": "Modified",
                    "ModifiedColumn": col,
                    "OldValue": prev_row.get(col),
                    "NewValue": curr_row.get(col),
                    "WhenModified": when_modified
                })

    return pd.DataFrame(results)

# =========================
# MAIN EXECUTION
# =========================

def main():

    os.makedirs("output", exist_ok=True)

    token = get_auth_token()
    headers = {"Authorization": f"Bearer {token}"}

    prev_df = load_report(PREVIOUS_FILE)
    curr_df = load_report(CURRENT_FILE)

    curr_df = fix_blank_registrations(curr_df, headers)

    status_df = fetch_status_dataset(headers)

    comparison_df = compare_reports(prev_df, curr_df, status_df)

    comparison_df.to_excel(OUTPUT_FILE, index=False)

    print("Comparison completed successfully.")

if __name__ == "__main__":
    main()
